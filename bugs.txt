BUG 4 — client_reference_id fix:
The sessionId is undefined when Stripe 
redirect fires because save-property 
API call is not awaited properly.

Fix payAndLoad() definitively:
async function payAndLoad(tier, price) {
  if(!selProp) return
  
  showStripeOverlay()
  
  let sessionId = 'unknown'
  try {
    const res = await fetch('/api/save-property', {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        property: selProp, 
        tier: tier
      })
    })
    const data = await res.json()
    sessionId = data.sessionId || 'unknown'
    console.log('SessionId confirmed:', sessionId)
  } catch(e) {
    console.error('Save property failed:', e)
  }
  
  localStorage.setItem('epc2c_prop', 
    JSON.stringify(selProp))
  localStorage.setItem('epc2c_tier', tier)
  localStorage.setItem('epc2c_session', sessionId)
  
  await new Promise(r => setTimeout(r, 2000))
  
  const stripeUrl = LINKS[tier] + 
    '?client_reference_id=' + 
    encodeURIComponent(sessionId)
  
  console.log('Redirecting to:', stripeUrl)
  window.location.href = stripeUrl
}
